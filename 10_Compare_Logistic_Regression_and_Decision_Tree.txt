import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score, classification_report, confusion_matrix
from sklearn.tree import DecisionTreeClassifier, plot_tree

df = pd.read_csv("/home/backup/ml_datasets/adult_income.csv")
print(df.head())
print(df.isnull().sum())

X = df.drop("class", axis=1)
y = df["class"].map({"<=50K": 0, ">50K": 1})
print(y.isnull().sum())
print(y.value_counts())

numerical_features = X.select_dtypes(include=["int64", "float64"]).columns.tolist()
categorical_features = X.select_dtypes(include=["category", "object"]).columns.tolist()

for column in numerical_features:
    X[column] = X[column].fillna(X[column].median())
for column in categorical_features:
    X[column] = X[column].fillna(X[column].mode()[0])

encoder = OneHotEncoder(handle_unknown="ignore", sparse_output=False)
X_categorical = encoder.fit_transform(X[categorical_features])
encoded_feature_names = encoder.get_feature_names_out(categorical_features)
X_categorical = pd.DataFrame(X_categorical, columns=encoded_feature_names, index=X.index)

scaler = StandardScaler()
X_numerical = scaler.fit_transform(X[numerical_features])
X_numerical = pd.DataFrame(X_numerical, columns=numerical_features, index=X.index)
X_processed = pd.concat([X_numerical, X_categorical], axis=1)
print("Final processed dataset shape:", X_processed.shape)

X_train, X_test, y_train, y_test = train_test_split(
    X_processed, y, test_size=0.2, random_state=42, stratify=y
)
print(f"Training set size: {X_train.shape[0]}")
print(f"Testing set size: {X_test.shape[0]}")

model = LogisticRegression(max_iter=2000)
model.fit(X_train, y_train)
y_pred_log = model.predict(X_test)

print("Training Decision Tree....")
tree_model = DecisionTreeClassifier(criterion="entropy", max_depth=4, random_state=42)
tree_model.fit(X_train, y_train)
y_pred_tree = tree_model.predict(X_test)

print("Logistic Regression:")
print(classification_report(y_test, y_pred_log, zero_division=0))
print("\nDecision Tree:")
print(classification_report(y_test, y_pred_tree, zero_division=0))

results = pd.DataFrame({
    "Model": ["Logistic Regression", "Decision Tree"],
    "Accuracy": [accuracy_score(y_test, y_pred_log), accuracy_score(y_test, y_pred_tree)],
    "Precision": [precision_score(y_test, y_pred_log, zero_division=0), precision_score(y_test, y_pred_tree, zero_division=0)],
    "Recall": [recall_score(y_test, y_pred_log, zero_division=0), recall_score(y_test, y_pred_tree, zero_division=0)],
    "F1 Score": [f1_score(y_test, y_pred_log, zero_division=0), f1_score(y_test, y_pred_tree, zero_division=0)],
})
print(results)
print("\nConfusion Matrices:")
print("\nLogistic Regression:")
print(confusion_matrix(y_test, y_pred_log))
print("\nDecision Tree:")
print(confusion_matrix(y_test, y_pred_tree))

plt.figure(figsize=(20, 10))
plot_tree(
    tree_model,
    feature_names=X_processed.columns,
    class_names=["<=50K", ">50K"],
    filled=True,
    rounded=True,
    max_depth=3,
    fontsize=8,
)
plt.title("Decision Tree for Adult Income Prediction")
plt.show()
