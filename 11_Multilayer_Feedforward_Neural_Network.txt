import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.neural_network import MLPClassifier
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score

data = pd.read_csv("/home/backup/ml_datasets/wine_quality_red.csv")
print("Dataset Shape:", data.shape)
print("\nFirst 5 rows")
print(data.head())

X = data.drop("quality", axis=1)
y = data["quality"]
y = (y > 6).astype(int)
print("\nClass Distributions:")
print(y.value_counts())

X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.20, random_state=42, stratify=y
)

scaler = StandardScaler()
X_train = scaler.fit_transform(X_train)
X_test = scaler.transform(X_test)

models = {
    "1 Hidden Layer - 10 Neurons": (10,),
    "1 Hidden Layer - 50 Neurons": (50,),
    "2 Hidden Layer - 50,25 Neurons": (50, 25),
    "2 Hidden Layer - 100,50 Neurons": (100, 50),
    "3 Hidden Layer - 100,50,25 Neurons": (100, 50, 25),
    "3 Hidden Layer - 100,100,50 Neurons": (100, 100, 50),
}

results = []
for name, architecture in models.items():
    print("\nTraining:", name)
    mlp = MLPClassifier(
        hidden_layer_sizes=architecture,
        activation="relu",
        solver="adam",
        max_iter=1000,
        random_state=42,
    )
    mlp.fit(X_train, y_train)
    y_pred = mlp.predict(X_test)
    accuracy = accuracy_score(y_test, y_pred)
    precision = precision_score(y_test, y_pred, zero_division=0)
    recall = recall_score(y_test, y_pred, zero_division=0)
    f1 = f1_score(y_test, y_pred, zero_division=0)
    results.append([name, accuracy, precision, recall, f1])

result_df = pd.DataFrame(
    results,
    columns=["Architecture", "Accuracy", "Precision", "Recall", "F1 Score"],
)
print("\n\nPerformance Comparison:")
print(result_df.to_string(index=False))
best_model = result_df.loc[result_df["F1 Score"].idxmax()]
print("Best Architecture:")
print(best_model)
